### 4.3 Metrics Dashboard (Portfolio Website)

**Live ROI Calculator** (Interactive Widget):
```typescript
// components/ROICalculator.tsx
export default function ROICalculator() {
  const [manualHours, setManualHours] = useState(15);
  const [hourlyRate, setHourlyRate] = useState(100);
  const [automationHours, setAutomationHours] = useState(40);

  const monthlySavings = manualHours * hourlyRate;
  const annualSavings = monthlySavings * 12;
  const investment = automationHours * hourlyRate;
  const roi = ((annualSavings - investment) / investment * 100).toFixed(0);

  return (
    <div className="roi-calculator">
      <h3>Calculate Your Automation ROI</h3>

      <label>
        Hours spent on manual tasks/month:
        <input
          type="range"
          min="5"
          max="40"
          value={manualHours}
          onChange={(e) => setManualHours(Number(e.target.value))}
        />
        {manualHours} hours
      </label>

      <label>
        Your hourly rate:
        <input
          type="number"
          value={hourlyRate}
          onChange={(e) => setHourlyRate(Number(e.target.value))}
        />
        ${hourlyRate}/hour
      </label>

      <label>
        Hours to build automation:
        <input
          type="range"
          min="10"
          max="100"
          value={automationHours}
          onChange={(e) => setAutomationHours(Number(e.target.value))}
        />
        {automationHours} hours
      </label>

      <div className="results">
        <h4>Your ROI:</h4>
        <p className="highlight">{roi}%</p>
        <p>Annual savings: ${annualSavings.toLocaleString()}</p>
        <p>Investment: ${investment.toLocaleString()}</p>
        <p>Payback period: {(investment / monthlySavings).toFixed(1)} months</p>
      </div>

      <button>Book a Consultation</button>
    </div>
  );
}
````
